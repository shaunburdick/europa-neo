  ```bash
  # ADDED comment lines (full-line //, /*, * AND trailing //)
  git diff -U0 main...HEAD -- packages | awk '
    /^\+\+\+/ { if ($0 ~ /^\+\+\+ b\//) f = substr($0, 7); next }
    /^\+/ {
      line = substr($0, 2)
      if (line ~ /^[[:space:]]*(\/\/|\/\*|\*)/ || line ~ /[[:space:]]\/\/[^\/]/) print "ADD", f
    }' | sort | uniq -c
  # REMOVED comment lines (path from the --- a/ header)
  git diff -U0 main...HEAD -- packages | awk '
    /^--- a\// { f = substr($0, 7); next }
    /^-/ && !/^---/ {
      line = substr($0, 2)
      if (line ~ /^[[:space:]]*(\/\/|\/\*|\*)/ || line ~ /[[:space:]]\/\/[^\/]/) print "DEL", f
    }' | sort | uniq -c
  ```
